# Airlock Core Implementation Plan

**Goal:** The pieces every agent shares: the packet, the desk checks, the fixed chat copy, the Switch client and the mock bank.
**Architecture:** Plain TypeScript modules with no framework. Logic takes a small client interface, so tests inject fakes and need no network beyond fixed local ports.
**Tech Stack:** Node >=22.18 (type stripping, `node:test`), TypeScript 5.9.3, @types/node 22.20.5, ws 8.22.1 and @types/ws 8.18.2 (test server only)
**Source spec:** docs/superpowers/specs/2026-11-04-invoice-airlock-design.md

**Where this runs:** on the VPS, as the OS user `airlock-build`, from `~/invoice-airlock`. That user has no sudo and no access to the main Docker daemon. Its rootless Docker is the only Docker it can use.

**House rules for every task here:**

- Node runs TypeScript directly through type stripping, so use only erasable syntax (no enums, no parameter properties), and write imports with the `.ts` extension. Type-only imports use `import type`.
- There are no runtime npm dependencies. Dev dependencies are only `typescript`, `@types/node`, `@types/ws` and `ws`.
- Test servers bind fixed ports in 24000-24999, never port 0, because this user's firewall refuses every other loopback port.
- Never use `as any`. Secrets appear only as environment variable names.
- Nothing here moves real money. The bank is a mock, every vendor is synthetic, and every secret a stack uses is generated per run and thrown away with it.
- Docker-gated tasks start a throwaway local stack on this user's rootless daemon. The user approved that at the design gate, and it is not a deploy, so those tasks are not human-gated. The only human-gated tasks are 5.3 and 7.5.
- `.gitignore` from task 1.1 covers `.run/`, `proof-output/`, `.taste-skill/` and `node_modules/`, so running a stack leaves nothing to commit from those.

`src/packet/schema.ts`

```ts
// The only thing that crosses from ap-intake to ap-release. No field can hold prose.
export type Packet = {
  invoice_no: string;
  vendor_id: string;
  po_no: string;
  grn_no: string;
  amount_cents: number;
  currency: 'USD';
  due_date: string;
  routing: string;
  account: string;
  source_sha256: string;
};

export type ParseResult =
  | { ok: true; packet: Packet }
  | { ok: false; reason: string };

export const PACKET_PREFIX = 'AIRLOCK/1 ';
export const PACKET_ATTEMPT_PREFIX = 'AIRLOCK/';
export const MAX_PACKET_BYTES = 1024;
const MAX_AMOUNT_CENTS = 10_000_000_000;

const STRING_RULES: Record<string, RegExp> = {
  invoice_no: /^INV-\d{1,10}$/,
  vendor_id: /^V-\d{4}$/,
  po_no: /^PO-\d{4,8}$/,
  grn_no: /^GRN-\d{4,8}$/,
  due_date: /^\d{4}-\d{2}-\d{2}$/,
  routing: /^\d{9}$/,
  account: /^\d{6,17}$/,
  source_sha256: /^[0-9a-f]{64}$/,
};

const ALL_KEYS = [...Object.keys(STRING_RULES), 'amount_cents', 'currency'].sort();

function isRealDate(value: string): boolean {
  const parsed = new Date(`${value}T00:00:00Z`);
  return !Number.isNaN(parsed.getTime()) && parsed.toISOString().slice(0, 10) === value;
}

export function validatePacket(value: unknown): ParseResult {
  if (typeof value !== 'object' || value === null || Array.isArray(value)) {
    return { ok: false, reason: 'not an object' };
  }
  const record = value as Record<string, unknown>;
  const keys = Object.keys(record).sort();
  if (keys.length !== ALL_KEYS.length || keys.some((key, i) => key !== ALL_KEYS[i])) {
    return { ok: false, reason: 'wrong keys' };
  }
  for (const [key, rule] of Object.entries(STRING_RULES)) {
    const field = record[key];
    if (typeof field !== 'string' || !rule.test(field)) {
      return { ok: false, reason: `bad ${key}` };
    }
  }
  if (!isRealDate(record['due_date'] as string)) {
    return { ok: false, reason: 'bad due_date' };
  }
  const amount = record['amount_cents'];
  if (typeof amount !== 'number' || !Number.isInteger(amount) || amount < 1 || amount > MAX_AMOUNT_CENTS) {
    return { ok: false, reason: 'bad amount_cents' };
  }
  if (record['currency'] !== 'USD') {
    return { ok: false, reason: 'bad currency' };
  }
  return { ok: true, packet: record as unknown as Packet };
}

export function parsePacket(line: string): ParseResult {
  if (Buffer.byteLength(line, 'utf8') > MAX_PACKET_BYTES) {
    return { ok: false, reason: 'too long' };
  }
  if (!line.startsWith(PACKET_PREFIX) || line.includes('\n')) {
    return { ok: false, reason: 'not a packet line' };
  }
  let value: unknown;
  try {
    value = JSON.parse(line.slice(PACKET_PREFIX.length));
  } catch {
    return { ok: false, reason: 'not JSON' };
  }
  return validatePacket(value);
}

export function formatPacket(packet: Packet): string {
  const ordered = {
    invoice_no: packet.invoice_no,
    vendor_id: packet.vendor_id,
    po_no: packet.po_no,
    grn_no: packet.grn_no,
    amount_cents: packet.amount_cents,
    currency: packet.currency,
    due_date: packet.due_date,
    routing: packet.routing,
    account: packet.account,
    source_sha256: packet.source_sha256,
  };
  return `${PACKET_PREFIX}${JSON.stringify(ordered)}`;
}
```

export type Vendor = { vendor_id: string; name: string; routing: string; account: string };

export type Desk = OpenOrders & { vendors: Vendor[] };

function readJson<T>(path: string): T {
  return JSON.parse(readFileSync(path, 'utf8')) as T;
}

export function loadOpenOrders(dir: string): OpenOrders {
  return {
    purchaseOrders: readJson<PurchaseOrder[]>(join(dir, 'purchase-orders.json')),
    goodsReceipts: readJson<GoodsReceipt[]>(join(dir, 'goods-receipts.json')),

}

export function loadDesk(dir: string): Desk {
  return { ...loadOpenOrders(dir), vendors: readJson<Vendor[]>(join(dir, 'vendors.json')) };

`src/desk/check.ts`

```ts
import type { Packet } from '../packet/schema.ts';
import type { Desk, Vendor } from './data.ts';

export type HoldReason =
  | 'unknown-vendor'
  | 'bank-mismatch'
  | 'po-mismatch'
  | 'grn-missing'
  | 'amount-mismatch';

export type Decision =
  | { kind: 'duplicate'; invoiceNo: string }
  | { kind: 'held'; reason: HoldReason; packet: Packet; vendor: Vendor | null }
  | { kind: 'cleared'; packet: Packet; vendor: Vendor };

// The first failure decides. The order is part of the design.
export function checkPacket(packet: Packet, desk: Desk, decided: ReadonlySet<string>): Decision {
  if (decided.has(packet.invoice_no)) {
    return { kind: 'duplicate', invoiceNo: packet.invoice_no };
  }
  const vendor = desk.vendors.find((v) => v.vendor_id === packet.vendor_id);
  if (!vendor) {
    return { kind: 'held', reason: 'unknown-vendor', packet, vendor: null };
  }
  if (vendor.routing !== packet.routing || vendor.account !== packet.account) {
    return { kind: 'held', reason: 'bank-mismatch', packet, vendor };
  }
  const po = desk.purchaseOrders.find((p) => p.po_no === packet.po_no && p.vendor_id === vendor.vendor_id);
  if (!po) {
    return { kind: 'held', reason: 'po-mismatch', packet, vendor };
  }
  const grn = desk.goodsReceipts.find((g) => g.grn_no === packet.grn_no && g.po_no === po.po_no);
  if (!grn) {
    return { kind: 'held', reason: 'grn-missing', packet, vendor };
  }
  if (packet.amount_cents !== grn.received_amount_cents || packet.amount_cents > po.amount_cents) {
    return { kind: 'held', reason: 'amount-mismatch', packet, vendor };
  }
  return { kind: 'cleared', packet, vendor };
}
```

    case 'unknown-vendor':
      return `Vendor ${packet.vendor_id} is not in the vendor master. Amount ${amount}.`;
    case 'bank-mismatch':
      return `Bank details differ from the vendor master for ${vendor?.name ?? packet.vendor_id}: account ending ${lastFour(packet.account)}, on file ending ${lastFour(vendor?.account ?? '')}. Amount ${amount}.`;
    case 'po-mismatch':
      return `${packet.po_no} does not belong to ${vendor?.name ?? packet.vendor_id}. Amount ${amount}.`;
    case 'grn-missing':
      return `No goods receipt ${packet.grn_no} for ${packet.po_no}. Amount ${amount}.`;
    case 'amount-mismatch':
      return `The amount does not match goods receipt ${packet.grn_no}. Amount ${amount}.`;
  }
}

    `**Held: ${packet.invoice_no}**`,
    holdSentence(packet, reason, vendor),
    'This cannot be paid in chat. Call the vendor on the number on file.',
    codeBlock(`AIRLOCK-HELD/1 ${JSON.stringify({ invoice_no: packet.invoice_no, reason })}`),
  ].join('\n');
}

  const match = /^AIRLOCK-(CLEARED|HELD)\/1 (\{.*\})$/m.exec(body);
  if (!match) return null;
  let value: unknown;
  try {
    value = JSON.parse(match[2] as string);
  } catch {
    return null;
  }
  if (typeof value !== 'object' || value === null) return null;
  const record = value as Record<string, unknown>;
  if (typeof record['invoice_no'] !== 'string') return null;
  if (match[1] === 'HELD') {
    if (typeof record['reason'] !== 'string') return null;
    return { kind: 'held', invoice_no: record['invoice_no'], reason: record['reason'] };

    typeof record['vendor_id'] !== 'string' ||
    typeof record['vendor_name'] !== 'string' ||
    typeof record['amount_cents'] !== 'number' ||
    typeof record['routing'] !== 'string' ||
    typeof record['account'] !== 'string'
  ) {
    return null;
  }
  return {
    kind: 'cleared',
    invoice_no: record['invoice_no'],
    vendor_id: record['vendor_id'],
    vendor_name: record['vendor_name'],
    amount_cents: record['amount_cents'],
    routing: record['routing'],
    account: record['account'],
  };
}

export type Attachment = {
  filename: string;
  mimetype: string;
  size: number;
  mxc: string;
  msgtype: string;
};

export type MessageFrame = {
  kind: 'message';
  roomId: string;
  messageId: string;
  addressed: boolean;
  sender: string;
  senderName: string;
  senderKind: string | null;
  body: string;
  timestamp: number;
  threadId: string | null;
  attachments: Attachment[];
};

export type RoomJoinFrame = {
  kind: 'room_join';
  roomId: string;
  member: string;
  memberName: string;
  timestamp: number;
  listening: boolean;
};

export type Frame =
  | { kind: 'ping' }
  | { kind: 'connection_state' }
  | { kind: 'evicted'; reason: string }
  | { kind: 'refused'; detail: string }
  | { kind: 'other'; name: string }
  | MessageFrame
  | RoomJoinFrame;

export type ParsedFrame = { frame: Frame; cursor: number | null };

function asRecord(value: unknown): Record<string, unknown> {
  return typeof value === 'object' && value !== null ? (value as Record<string, unknown>) : {};
}

function asString(value: unknown): string {
  return typeof value === 'string' ? value : '';
}

export function parseFrame(raw: string): ParsedFrame {
  const outer = asRecord(JSON.parse(raw));
  const name = asString(outer['event']);
  const data = asRecord(outer['data']);
  const cursor = typeof outer['id'] === 'number' ? outer['id'] : null;
  if (name === 'ping' || outer['type'] === 'ping') {
    return { frame: { kind: 'ping' }, cursor };
  }
  if (name === 'connection_state') {
    return { frame: { kind: 'connection_state' }, cursor };
  }
  if (name === 'evicted') {
    return { frame: { kind: 'evicted', reason: asString(data['reason']) }, cursor };
  }
  if (name === 'refused') {
    return { frame: { kind: 'refused', detail: JSON.stringify(outer['data'] ?? null) }, cursor };
  }
  const payload = asRecord(data['payload']);
  const roomId = asString(data['room_id']);
  if (name === 'message') {
    return {
      frame: {
        kind: 'message',
        roomId,
        messageId: asString(payload['message_id']),
        addressed: payload['addressed'] === true,
        sender: asString(payload['sender']),
        senderName: asString(payload['sender_name']),
        senderKind: typeof payload['sender_kind'] === 'string' ? payload['sender_kind'] : null,
        body: asString(payload['body']),
        timestamp: typeof payload['timestamp'] === 'number' ? payload['timestamp'] : 0,
        threadId: typeof payload['thread_id'] === 'string' ? payload['thread_id'] : null,
        attachments: Array.isArray(payload['attachments']) ? (payload['attachments'] as Attachment[]) : [],
      },
      cursor,
    };
  }
  if (name === 'room_join') {
    return {
      frame: {
        kind: 'room_join',
        roomId,
        member: asString(payload['member']),
        memberName: asString(payload['member_name']),
        timestamp: typeof payload['timestamp'] === 'number' ? payload['timestamp'] : 0,
        listening: payload['listening'] === true,
      },
      cursor,
    };
  }
  return { frame: { kind: 'other', name }, cursor };
}

export type OpResult = { status: number; body: unknown; retryAfter: string | null };

// What agent logic needs from Switch. Tests inject a fake.
export interface DeskClient {
  op(name: string, args?: Record<string, unknown>): Promise<OpResult>;
  say(roomId: string, content: string): Promise<OpResult>;
}

export interface ReaderClient extends DeskClient {
  media(roomId: string, mxc: string): Promise<Buffer>;
}

export type AgentIdentity = { id: string; apiKey: string; connectionId: string; name: string };
export type ClientOptions = {
  baseUrl: string;
  identity: AgentIdentity;
  filter: 'all' | 'addressed';
  retryDelayMs?: number;
  log?: (line: string) => void;
};

export function resultOf<T>(result: OpResult): T | null {
  if (result.status !== 200 || typeof result.body !== 'object' || result.body === null) return null;
  const inner = (result.body as Record<string, unknown>)['result'];
  return inner === undefined ? null : (inner as T);
}

const MAX_BACKOFF_MS = 15_000;
const CONFLICT_RETRIES = 5;

async function readBody(response: Response): Promise<unknown> {
  const text = await response.text();
  try {
    return JSON.parse(text);
  } catch {
    return text;
  }
}

export class AgentClient implements ReaderClient {
  private readonly options: ClientOptions;
  private socket: WebSocket | null = null;
  private stopped = false;
  private cursor = 0;
  private backoffMs = 1000;
  private onFatal: (reason: string) => void = () => {};

  constructor(options: ClientOptions) {
    this.options = options;
  }

  // Resolves at the first connection_state. Fatal means evicted by a takeover, which must not reconnect.
  start(onFrame: (frame: Frame) => void | Promise<void>, onFatal: (reason: string) => void): Promise<void> {
    this.onFatal = onFatal;
    return new Promise((resolve) => this.open(onFrame, resolve));
  }

  stop(): void {
    this.stopped = true;
    this.socket?.close();
  }

  private log(line: string): void {
    this.options.log?.(`[${this.options.identity.name}] ${line}`);
  }

  private open(onFrame: (frame: Frame) => void | Promise<void>, ready: () => void): void {
    const { baseUrl, identity, filter } = this.options;
    const query = new URLSearchParams({
      connection_id: identity.connectionId,
      filter,
      scope: 'all',
      client: 'invoice-airlock',
      client_version: '1',
    });
    const url = `${baseUrl.replace(/^http/, 'ws')}/agents/${identity.id}/connection/ws?${query.toString()}`;
    const socket = new WebSocket(url, { headers: { Authorization: `Bearer ${identity.apiKey}` } });
    this.socket = socket;
    socket.onmessage = (event) => {
      const { frame, cursor } = parseFrame(String(event.data));
      if (frame.kind === 'ping') {
        // Answered before anything else. Six seconds of silence evicts the connection.
        socket.send(JSON.stringify({ type: 'pong', cursor: this.cursor }));
        return;
      }
      if (cursor !== null) this.cursor = cursor;
      if (frame.kind === 'connection_state') {
        this.backoffMs = 1000;
        ready();
      }
      if (frame.kind === 'evicted' && frame.reason === 'taken_over') {
        this.stopped = true;
        this.onFatal('taken_over');
        return;
      }
      Promise.resolve(onFrame(frame)).catch((error: unknown) => this.log(`handler failed: ${String(error)}`));
    };
    socket.onerror = () => this.log('socket error');
    socket.onclose = () => {
      if (this.stopped) return;
      const delay = this.backoffMs;
      this.backoffMs = Math.min(this.backoffMs * 2, MAX_BACKOFF_MS);
      setTimeout(() => this.open(onFrame, ready), delay);
    };
  }

  private async post(path: string, body: unknown): Promise<OpResult> {
    const { baseUrl, identity } = this.options;
    try {
      const response = await fetch(`${baseUrl}${path}`, {
        method: 'POST',
        headers: {
          Authorization: `Bearer ${identity.apiKey}`,
          'Content-Type': 'application/json',
          'X-Switch-Connection-Id': identity.connectionId,
        },
        body: JSON.stringify(body),
      });
      return { status: response.status, body: await readBody(response), retryAfter: response.headers.get('retry-after') };
    } catch (error) {
      return { status: 0, body: String(error), retryAfter: null };
    }
  }

  // A refused call comes back as data. Only "connection is not open" is worth retrying.
  async op(name: string, args: Record<string, unknown> = {}): Promise<OpResult> {
    const path = `/agents/${this.options.identity.id}/ops/${name}`;
    let result = await this.post(path, args);
    for (let i = 0; i < CONFLICT_RETRIES && result.status === 409 && JSON.stringify(result.body).includes('is not open'); i++) {
      await new Promise((resolve) => setTimeout(resolve, this.options.retryDelayMs ?? 1000));
      result = await this.post(path, args);
    }
    return result;
  }

  say(roomId: string, content: string): Promise<OpResult> {
    return this.post(`/agents/${this.options.identity.id}/message`, { room_id: roomId, content });
  }

  async media(roomId: string, mxc: string): Promise<Buffer> {
    const { baseUrl, identity } = this.options;
    const query = new URLSearchParams({ mxc });
    const response = await fetch(`${baseUrl}/agents/${identity.id}/rooms/${roomId}/media?${query.toString()}`, {
      headers: { Authorization: `Bearer ${identity.apiKey}` },
    });
    if (response.status !== 200) {
      throw new Error(`media download returned ${response.status}`);
    }
    return Buffer.from(await response.arrayBuffer());
  }
}

import { resultOf, type OpResult } from './agent-client.ts';
import type { Attachment } from './frames.ts';

export type ContextEntry = {
  id: string;
  kind: string;
  sender: string;
  senderName: string;
  body: string;
  timestamp: number;
  attachments: Attachment[];
};

type RawEntry = {
  id?: string;
  kind?: string;
  sender?: string;
  sender_name?: string;
  body?: string;
  timestamp?: number;
  attachments?: Attachment[];
};

type RawContext = { threads?: { root?: RawEntry; replies?: RawEntry[] }[] };

function toEntry(raw: RawEntry): ContextEntry {
  return {
    id: raw.id ?? '',
    kind: raw.kind ?? 'message',
    sender: raw.sender ?? '',
    senderName: raw.sender_name ?? '',
    body: raw.body ?? '',
    timestamp: raw.timestamp ?? 0,
    attachments: raw.attachments ?? [],
  };
}

// read_context groups a room into threads. Agents here want a flat timeline, oldest first.
export function flattenContext(result: OpResult): ContextEntry[] {
  const context = resultOf<RawContext>(result);
  if (!context) return [];
  const entries: ContextEntry[] = [];
  for (const thread of context.threads ?? []) {
    if (thread.root) entries.push(toEntry(thread.root));
    for (const reply of thread.replies ?? []) entries.push(toEntry(reply));
  }
  return entries.sort((a, b) => a.timestamp - b.timestamp);
}

export type PaymentRequest = {
  invoice_no: string;
  vendor_id: string;
  vendor_name: string;
  amount_cents: number;
  routing: string;
  account: string;
  released_by: string;
};

export type LedgerEntry = PaymentRequest & { entry: number; at: string };
export type LedgerView = { opening_balance_cents: number; balance_cents: number; entries: LedgerEntry[] };
export type PayResult = { ok: true; entry: LedgerEntry } | { ok: false; code: 'already-paid' | 'insufficient' };

export const OPENING_BALANCE_CENTS = 25_000_000;

export function parsePaymentRequest(value: unknown): PaymentRequest | null {
  if (typeof value !== 'object' || value === null) return null;
  const record = value as Record<string, unknown>;
  for (const key of ['invoice_no', 'vendor_id', 'vendor_name', 'routing', 'account', 'released_by']) {
    if (typeof record[key] !== 'string' || record[key] === '') return null;
  }
  const amount = record['amount_cents'];
  if (typeof amount !== 'number' || !Number.isInteger(amount) || amount < 1) return null;
  return {
    invoice_no: record['invoice_no'] as string,
    vendor_id: record['vendor_id'] as string,
    vendor_name: record['vendor_name'] as string,
    amount_cents: amount,
    routing: record['routing'] as string,
    account: record['account'] as string,
    released_by: record['released_by'] as string,
  };
}

export class Ledger {
  private readonly opening: number;
  private readonly entries: LedgerEntry[] = [];

  constructor(opening: number = OPENING_BALANCE_CENTS) {
    this.opening = opening;
  }

  private balance(): number {
    return this.entries.reduce((total, entry) => total - entry.amount_cents, this.opening);
  }

  pay(request: PaymentRequest, at: Date = new Date()): PayResult {
    if (this.entries.some((entry) => entry.invoice_no === request.invoice_no)) {
      return { ok: false, code: 'already-paid' };
    }
    if (request.amount_cents > this.balance()) {
      return { ok: false, code: 'insufficient' };
    }
    const entry: LedgerEntry = { ...request, entry: this.entries.length + 1, at: at.toISOString() };
    this.entries.push(entry);
    return { ok: true, entry };
  }

function tokenMatches(request: IncomingMessage, token: string): boolean {
  const given = Buffer.from(request.headers.authorization ?? '');
  const wanted = Buffer.from(`Bearer ${token}`);
  return given.length === wanted.length && timingSafeEqual(given, wanted);
}

export function createBankServer(token: string, ledger: Ledger): Server {
  return createServer((request, response) => {
    if (request.method === 'GET' && request.url === '/') {
      response.writeHead(200, { 'content-type': 'text/html; charset=utf-8' });
      response.end(PAGE);
      return;
    }
    if (request.method === 'GET' && request.url === '/ledger.json') {
      send(response, 200, ledger.view());
      return;
    }
    if (request.method === 'POST' && request.url === '/payments') {
      if (!tokenMatches(request, token)) {
        send(response, 401, { error: 'unauthorized' });
        return;
      }
      let raw = '';
      request.on('data', (chunk) => { raw += String(chunk); });
      request.on('end', () => {
        let parsed: unknown = null;
        try {
          parsed = JSON.parse(raw);
        } catch {
          parsed = null;
        }
        const payment = parsePaymentRequest(parsed);
        if (!payment) {
          send(response, 400, { error: 'bad payment' });
          return;
        }
        const result = ledger.pay(payment);
        if (result.ok) {
          send(response, 201, result.entry);
        } else if (result.code === 'already-paid') {
          send(response, 409, { error: result.code });
        } else {
          send(response, 400, { error: result.code });
        }
      });
      return;
    }
    send(response, 404, { error: 'not found' });
  });
}

async function readBody(response: Response): Promise<unknown> {

  } catch {
    return text;

// A person's Gateway session. Login sets cookies, so this keeps a small jar.
export class GatewaySession {
  private readonly baseUrl: string;
  private readonly cookies = new Map<string, string>();

  constructor(baseUrl: string) {
    this.baseUrl = baseUrl.replace(/\/$/, '');
  }

  async login(email: string, password: string): Promise<HttpResult> {
    return this.request('POST', '/auth/login', { email, password });
  }

  async request(method: string, path: string, body?: unknown): Promise<HttpResult> {
    const headers: Record<string, string> = {};
    if (body !== undefined) headers['Content-Type'] = 'application/json';
    if (this.cookies.size > 0) {
      headers['Cookie'] = [...this.cookies].map(([name, value]) => `${name}=${value}`).join('; ');
    }
    const response = await fetch(`${this.baseUrl}${path}`, {
      method,
      headers,
      ...(body !== undefined ? { body: JSON.stringify(body) } : {}),
    });
    for (const cookie of response.headers.getSetCookie()) {
      const pair = cookie.split(';')[0] ?? '';
      const at = pair.indexOf('=');
      if (at > 0) this.cookies.set(pair.slice(0, at), pair.slice(at + 1));
    }
    return { status: response.status, body: await readBody(response) };
  }
}

export function must(result: HttpResult, label: string): Record<string, unknown> {
  if (result.status < 200 || result.status >= 300) {
    throw new Error(`${label} returned ${result.status}: ${JSON.stringify(result.body).slice(0, 300)}`);
  }
  return typeof result.body === 'object' && result.body !== null ? (result.body as Record<string, unknown>) : {};
}

export type Post = {

  rootId: string;
  username: string;
  message: string;
  type: string;
  createAt: number;

type RawPosts = { order: string[]; posts: Record<string, RawPost> };

async function readBody(response: Response): Promise<unknown> {
  const text = await response.text();
  try {
    return JSON.parse(text);
  } catch {
    return text;
  }
}

// One person's view of Mattermost, so a transcript shows what that person can see.
export class MattermostSession {
  private readonly baseUrl: string;
  private readonly usernames = new Map<string, string>();
  private token = '';
  userId = '';

  constructor(baseUrl: string) {
    this.baseUrl = baseUrl.replace(/\/$/, '');
  }

  async login(username: string, password: string): Promise<boolean> {
    const response = await fetch(`${this.baseUrl}/users/login`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ login_id: username, password }),
    });
    if (response.status !== 200) return false;
    this.token = response.headers.get('Token') ?? '';
    this.userId = ((await response.json()) as { id: string }).id;
    return this.token !== '';
  }

  async request(method: string, path: string, body?: unknown): Promise<HttpResult> {
    const headers: Record<string, string> = { Authorization: `Bearer ${this.token}` };
    if (body !== undefined) headers['Content-Type'] = 'application/json';
    const response = await fetch(`${this.baseUrl}${path}`, {
      method,
      headers,
      ...(body !== undefined ? { body: JSON.stringify(body) } : {}),
    });
    return { status: response.status, body: await readBody(response) };
  }

  async channelId(team: string, channel: string): Promise<string | null> {
    const result = await this.request('GET', `/teams/name/${team}/channels/name/${channel}`);
    return result.status === 200 ? (result.body as { id: string }).id : null;
  }

  async postMessage(channelId: string, message: string, fileIds: string[] = []): Promise<string> {
    const result = await this.request('POST', '/posts', { channel_id: channelId, message, file_ids: fileIds });
    if (result.status !== 201) throw new Error(`post returned ${result.status}`);
    return (result.body as { id: string }).id;
  }

  async uploadAndPost(channelId: string, message: string, filename: string, bytes: Buffer): Promise<string> {
    const form = new FormData();
    form.append('channel_id', channelId);
    form.append('files', new Blob([new Uint8Array(bytes)], { type: 'application/pdf' }), filename);
    const response = await fetch(`${this.baseUrl}/files`, {
      method: 'POST',
      headers: { Authorization: `Bearer ${this.token}` },
      body: form,
    });
    if (response.status !== 201) throw new Error(`upload returned ${response.status}`);
    const uploaded = (await response.json()) as { file_infos: { id: string }[] };
    return this.postMessage(channelId, message, uploaded.file_infos.map((file) => file.id));
  }

  private async username(userId: string): Promise<string> {
    const known = this.usernames.get(userId);
    if (known !== undefined) return known;
    const result = await this.request('GET', `/users/${userId}`);
    const name = result.status === 200 ? (result.body as { username: string }).username : userId;
    this.usernames.set(userId, name);
    return name;
  }

  async posts(channelId: string): Promise<Post[]> {
    const result = await this.request('GET', `/channels/${channelId}/posts?per_page=200`);
    if (result.status !== 200) return [];
    const raw = result.body as RawPosts;
    const posts: Post[] = [];
    for (const id of [...raw.order].reverse()) {
      const post = raw.posts[id];
      if (!post) continue;
      posts.push({
        id: post.id,
        rootId: post.root_id ?? '',
        username: await this.username(post.user_id),
        message: post.message,
        type: post.type ?? '',
        createAt: post.create_at,
      });
    }
    return posts;
  }

  async clientConfig(): Promise<Record<string, string>> {
    const result = await this.request('GET', '/config/client?format=old');
    return result.status === 200 ? (result.body as Record<string, string>) : {};
  }
}
